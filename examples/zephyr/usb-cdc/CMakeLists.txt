cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(usb-cdc)
# sample_usbd_init_device() helper lives in the Zephyr tree (used read-only, not copied)

target_include_directories(app PRIVATE ${SAMPLE_USBD_DIR})
target_sources(app PRIVATE src/main.c ${SAMPLE_USBD_DIR}/sample_usbd_init.c)
